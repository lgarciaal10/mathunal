\textbf{1.} Sea $\vec{v} = \begin{bmatrix}1\\0\\-1\end{bmatrix}$ y $T:\mathbb{R}^3 \to \mathbb{R}^3$ lineal definida por $T(\vec{u}) = \operatorname{Proy}_{\vec{v}}(\vec{u})$.

\textbf{2.} Determine si las siguientes afirmaciones son verdaderas o falsas.

\begin{enumerate}[label=\alph*)]
\item $T:\mathbb{R}^3 \to \mathbb{R}^3$ que proyecta un vector $\vec{v}$ sobre $\begin{bmatrix}1\\1\\1\end{bmatrix}$ es invertible. \hfill \mbox{$\boxed{V}$ $\boxed{F}$}

\vspace{0.5cm}
\item La matriz $A = \begin{bmatrix}2&2\\1&1\end{bmatrix}\begin{bmatrix}2&-4\\-1&3\end{bmatrix}$ es invertible. \hfill \mbox{$\boxed{V}$ $\boxed{F}$}

\vspace{0.5cm}
\item $\{2-x,\; x,\; x-x^2,\; 1-2x^2+3x^3\}$ es L.D. en $\mathcal{P}_3$. \hfill \mbox{$\boxed{V}$ $\boxed{F}$}

\vspace{0.5cm}
\item $\mathcal{M}_{2\times 2} = \operatorname{gen}\left\{ \begin{bmatrix}1&1\\0&1\end{bmatrix}, \begin{bmatrix}0&1\\1&0\end{bmatrix}, \begin{bmatrix}1&0\\1&1\end{bmatrix}, \begin{bmatrix}0&-1\\1&0\end{bmatrix} \right\}$. \hfill \mbox{$\boxed{V}$ $\boxed{F}$}

\vspace{0.5cm}
\item $\left\{ \begin{bmatrix}x\\y\end{bmatrix} \;\middle|\; y \geq 0 \right\}$ es subespacio. \hfill \mbox{$\boxed{V}$ $\boxed{F}$}
\end{enumerate}

\examfooter

\textbf{3.} Sea $T:\mathbb{R}^3 \to \mathbb{R}^4$ lineal definida por $T\begin{pmatrix}a\\b\\c\end{pmatrix} = \begin{pmatrix}0\\a\\b/2\\c/3\end{pmatrix}$ y $S:\mathbb{R}^4 \to \mathbb{R}^3$ lineal definida por $S\begin{pmatrix}a\\b\\c\\d\end{pmatrix} = \begin{pmatrix}b\\2c\\3d\end{pmatrix}$.

Calcule $T \circ S\begin{pmatrix}2\\5\\-1\\-3\end{pmatrix}$.

\vspace{2cm}

\examfooter

\textbf{4.} Sea $\mathcal{B} = \left\{ \begin{pmatrix}-1\\-1\\0\end{pmatrix}, \begin{pmatrix}1\\0\\-1\end{pmatrix} \right\}$ base para un subespacio $Z$.

Encontrar un vector $\vec{v}$ tal que $[\vec{v}]_{\mathcal{B}} = \begin{bmatrix}1\\1\end{bmatrix}$.

Halle una base $\mathcal{B}$ para $\mathcal{L}$.

\textbf{6.} Sea
\[
A = \begin{pmatrix} 1&0&0&0&2\\ 1&1&0&0&-1\\ 2&3&1&0&1\\ 2&1&-1&1&-3 \end{pmatrix}.
\]

\begin{enumerate}[label=\alph*)]
\item Halle $\dim(\operatorname{Ren}(A))$.
\item Halle $\operatorname{Nulidad}(A)$.
\end{enumerate}

\textbf{7.} Sea $W = \operatorname{gen}\left\{ \begin{pmatrix}2\\1\\1\end{pmatrix}, \begin{pmatrix}-1\\-1\\-2\end{pmatrix} \right\}$.

\begin{enumerate}[label=\alph*)]
\item Halle una base para $W$.
\item Calcule $\dim W$.
\end{enumerate}

\textbf{8.} Sea $T:\mathbb{R}^2 \to \mathbb{R}^2$ lineal tal que
\[
T\begin{pmatrix}1\\0\end{pmatrix} = \begin{pmatrix}5\\1\end{pmatrix}, \qquad T\begin{pmatrix}3\\-1\end{pmatrix} = \begin{pmatrix}1\\-1\end{pmatrix}.
\]

\begin{enumerate}[label=\alph*)]
\item Halle $T\begin{pmatrix}0\\1\end{pmatrix}$.
\item Halle $T\begin{pmatrix}x\\y\end{pmatrix}$.
\item Si $T$ es invertible, calcule $\det A$, donde $A = [T^{-1}]$.
\end{enumerate}
